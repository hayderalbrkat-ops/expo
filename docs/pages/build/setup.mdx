import React, { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, ScrollView, StyleSheet, Alert, Image } from 'react-native';
import * as ImagePicker from 'expo-image-picker';
import * as DocumentPicker from 'expo-document-picker';

export default function App(){
  const [role,setRole]=useState('customer');
  const [pages,setPages]=useState('100');
  const [binding,setBinding]=useState(false);
  const [file,setFile]=useState(null);
  const [photo,setPhoto]=useState(null);
  const totalPages=parseInt(pages)||0;
  const totalPrice=totalPages*25+(binding?5000:0);
  const sheets=Math.ceil(totalPages/2);
  const pickFile=async()=>{let r=await DocumentPicker.getDocumentAsync({type:'application/pdf'});if(r.assets)setFile(r.assets[0])};
  const pickImage=async()=>{let r=await ImagePicker.launchCameraAsync({quality:0.8});if(!r.canceled)setPhoto(r.assets[0])};
  if(role==='tech')return(<View style={s.container}><Text style={s.logo}>لوحة الفني 🖨️</Text><View style={s.card}><Text>الملف: {file?.name||'صورة'}</Text><Text>عدد الصفحات: {totalPages} ({sheets} ورقة)</Text><Text style={s.price}>{totalPrice} د.ع</Text></View><TouchableOpacity onPress={()=>setRole('customer')}><Text style={{color:'blue',textAlign:'center',marginTop:20}}>رجوع للزبون</Text></TouchableOpacity></View>);
  return(<ScrollView style={s.container}><Text style={s.logo}>اطبعلي 🧑‍🎓</Text><View style={s.card}><Text style={s.title}>1. ارفع ملف للطباعة</Text><TouchableOpacity style={s.btnO} onPress={pickFile}><Text>📁 رفع PDF</Text></TouchableOpacity>{file&&<Text>✅ {file.name}</Text>}<TextInput style={s.input} keyboardType="numeric" value={pages} onChangeText={setPages} placeholder="عدد الصفحات"/><Text>الطباعة وجهين: {sheets} ورقة</Text><TouchableOpacity onPress={()=>setBinding(!binding)}><Text>{binding?'✅ تجليد':'⬜ بدون تجليد'} +5000</Text></TouchableOpacity><Text style={s.price}>المجموع: {totalPrice} د.ع</Text></View><View style={s.card}><Text style={s.title}>2. تصوير معاملة</Text><TouchableOpacity style={s.btn} onPress={pickImage}><Text style={s.btnT}>📸 صور المعاملة</Text></TouchableOpacity>{photo&&<Image source={{uri:photo.uri}} style={{width:'100%',height:200,marginTop:10,borderRadius:10}}/>}</View><TouchableOpacity style={s.btnBig} onPress={()=>{Alert.alert('تم الطلب',`المجموع ${totalPrice} د.ع - الدفع عند الاستلام`); setRole('tech')}}><Text style={s.btnT}>اطلب الان - دفع عند التوصيل</Text></TouchableOpacity></ScrollView>);
}
const s=StyleSheet.create({container:{flex:1,padding:20,backgroundColor:'#f7f7ff'},logo:{fontSize:30,fontWeight:'bold',textAlign:'center',marginTop:40,color:'#4a00e0'},card:{backgroundColor:'white',padding:15,borderRadius:15,marginBottom:15},title:{fontWeight:'bold',marginBottom:10},input:{borderWidth:1,borderColor:'#ddd',borderRadius:10,padding:12,marginVertical:10},btn:{backgroundColor:'#4a00e0',padding:15,borderRadius:10,alignItems:'center'},btnBig:{backgroundColor:'#00c853',padding:18,borderRadius:15,alignItems:'center',marginBottom:30},btnO:{borderWidth:1,borderColor:'#4a00e0',padding:15,borderRadius:10,alignItems:'center'},btnT:{color:'white',fontWeight:'bold'},price:{fontSize:18,fontWeight:'bold',color:'#4a00e0',marginTop:10}});

}
